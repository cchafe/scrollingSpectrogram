 [Unit]
 Description=My Sample Service

 [Service]
 Environment=DISPLAY=:0.0

 Type=oneshot
 RemainAfterExit=true
 ExecStartPre=/bin/sleep 20
 ExecStart=/bin/sh /home/cc/f40xfce/button.sh

 [Install]
 WantedBy=default.target

systemctl --user daemon-reload
